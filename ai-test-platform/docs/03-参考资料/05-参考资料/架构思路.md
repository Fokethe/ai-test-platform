# 需求分析 Agent 的工作流
def requirement_analysis_node(state: State):
    # 1. 检索相似需求
    similar_reqs = retriever.similarity_search(state["content"])
    
    # 2. LLM 分析（带上下文）
    analysis = llm.invoke(requirement_prompt.format(
        current=state["content"],
        references=similar_reqs,
        standards="验收标准必须可量化"
    ))
    
    return {"requirements": analysis.requirements}

# 功能点拆解 Agent
def feature_decomposition_node(state: State):
    for req in state["requirements"]:
        # 调用拆解 Skill
        features = llm.invoke(feature_prompt.format(req=req))
        
        # 保存到 Graph 状态（供下游使用）
        state["feature_tree"].extend(features)
    
    return {"feature_tree": state["feature_tree"]}

@mcp.tool()
def generate_test_case(feature: str, technique: str) -> List[TestCase]:
    """
    根据功能点和测试技术生成用例

Agent 工作流：
python
def test_generation_node(state: State):
    test_cases = []
    
    for feature in state["feature_tree"]:
        # 检索历史相似功能的用例作为参考
        examples = retriever.get_similar_tests(feature)
        
        # 生成多维度用例
        for technique in ["等价类", "边界值", "场景法"]:
            cases = mcp_client.call("generate_test_case", 
                                  feature=feature, 
                                  technique=technique,
                                  examples=examples)
            test_cases.extend(cases)
    
    return {"test_cases": test_cases, "count": len(test_cases)}

# 人机协同节点
def human_review_node(state: State):
    # 发送到前端审核台
    send_to_ui(state["test_cases"])
    
    # 等待人工反馈（阻塞或异步）
    feedback = wait_for_human_input()
    
    if feedback["action"] == "approve":
        return "save_to_db"
    elif feedback["action"] == "regenerate":
        return "test_generation"  # 回滚到生成节点
    else:
        return "manual_edit"  # 人工修改后入库

四、关键实现细节
1. 长文档处理策略
需求文档通常 50+ 页，采用 Map-Reduce 模式：
 
Map: 按章节切分，并行提取子需求
 
Reduce: 汇总去重，建立需求间依赖关系
2. 质量保证机制
 
一致性检查：Agent 检查需求 ↔ 功能点 ↔ 用例的追溯关系
 
覆盖率分析：确保每个需求点都有对应测试场景
 
可执行性验证：提取测试步骤中的动作词，验证是否可自动化
